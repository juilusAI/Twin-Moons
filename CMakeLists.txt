cmake_minimum_required(VERSION 3.15)
project(DemonSlayerTwinMoons CXX)
set(CMAKE_CXX_STANDARD 17)
file(GLOB SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(DemonSlayer WIN32 ${SOURCES})
target_link_libraries(DemonSlayer PRIVATE gdi32 user32 winmm)
if(MSVC)
  target_compile_options(DemonSlayer PRIVATE /utf-8 /O2)
endif()
